n=int(input())
a = list(map(int, input().split()))
tree = []
for x in a:
    if len(tree) == 0:
        tree.append([x, -1, -1])
    else:
        i = 0
        while True:
            if x <= tree[i][0]:
                if tree[i][1] == -1:
                    tree[i][1] = len(tree)
                    tree.append([x, -1, -1])
                    break
                else:
                    i = tree[i][1]

            else:
                if tree[i][2] == -1:
                    tree[i][2] = len(tree)
                    tree.append([x, -1, -1])
                    break
                else:
                    i = tree[i][2]

height = 0
stack = [(0, 1)]

while len(stack) > 0:
    i, level = stack.pop()

    if level > height:
        height = level

    if tree[i][1] != -1:
        stack.append((tree[i][1], level + 1))

    if tree[i][2] != -1:
        stack.append((tree[i][2], level + 1))

print(height)
